KONFIGURASI VLAN + ROUTER-ON-A-STICK + DHCP
Sumber tabel: Tabel_IP_VLAN_Erza.xlsx (sheet "IP VLAN")
Topologi: V5.2.pts | Semua host: 144.124.192.0/24 (mask 255.255.255.0)

ASUMSI (sesuaikan bila dosen menentukan lain):
- Mask semua /24 : 255.255.255.0
- Gateway tiap VLAN = 144.124.192.254 (subinterface router)
- Pool DHCP VLAN 40 = 144.124.192.100 - 150 (di luar .1-.15 statis)
- Port Fa0/X di bawah CONTOH — ganti dengan nomor port di topologimu.
  Prinsip: 1 port akses = 1 VLAN. Port ke router = trunk.

==============================================================
1. SWITCH (misal Switch0)
==============================================================
enable
configure terminal
hostname SW-UTAMA

vlan 10
 name SERVER
vlan 20
 name STAFF
vlan 30
 name LAB-PRODI
vlan 40
 name MAHASISWA
exit

! --- Port akses VLAN 10 (contoh Fa0/1-4: SRV_01, SRV_02, STF_01, STF_02)
interface range fa0/1 - 4
 switchport mode access
 switchport access vlan 10
 no shutdown
exit

! --- Port akses VLAN 20 (contoh Fa0/5-7: SRV_03, STF_03, STF_04)
interface range fa0/5 - 7
 switchport mode access
 switchport access vlan 20
 no shutdown
exit

! --- Port akses VLAN 30 (contoh Fa0/8-15: LAB1_01..PRODI_02)
interface range fa0/8 - 15
 switchport mode access
 switchport access vlan 30
 no shutdown
exit

! --- Port akses VLAN 40 (contoh Fa0/16-18 ke AP / MHS_01-03)
interface range fa0/16 - 18
 switchport mode access
 switchport access vlan 40
 no shutdown
exit

! --- Trunk ke router (contoh Fa0/24; kalau ada Gi0/1 pakai itu)
interface fa0/24
 switchport mode trunk
 no shutdown
exit

end
write memory

! Verifikasi:
!   show vlan brief
!   show interfaces trunk

==============================================================
2. ROUTER (router-on-a-stick, contoh kabel ke G0/0)
==============================================================
enable
configure terminal
hostname R-UTAMA

interface g0/0
 no shutdown
exit

interface g0/0.10
 description Gateway-VLAN10-SERVER
 encapsulation dot1Q 10
 ip address 144.124.192.254 255.255.255.0
 no shutdown
exit

interface g0/0.20
 description Gateway-VLAN20-STAFF
 encapsulation dot1Q 20
 ip address 144.124.192.254 255.255.255.0
 no shutdown
exit

interface g0/0.30
 description Gateway-VLAN30-LAB-PRODI
 encapsulation dot1Q 30
 ip address 144.124.192.254 255.255.255.0
 no shutdown
exit

interface g0/0.40
 description Gateway-VLAN40-MAHASISWA
 encapsulation dot1Q 40
 ip address 144.124.192.254 255.255.255.0
 no shutdown
exit

! --- DHCP hanya untuk VLAN 40 (MHS_01-03), .1-.99 dikecualikan
ip dhcp excluded-address 144.124.192.1 144.124.192.99
ip dhcp pool VLAN40-MHS
 network 144.124.192.0 255.255.255.0
 default-router 144.124.192.254
 dns-server 8.8.8.8
exit

end
write memory

! Verifikasi:
!   show ip interface brief
!   show ip dhcp binding   (setelah MHS request IP)
